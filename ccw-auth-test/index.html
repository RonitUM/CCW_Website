// Short-lived auth diagnostic BACKEND (API only). Stores nothing, logs nothing, never returns the token,
// never uses a saved secret. Fixed tenant/client only (not a general proxy). Needs an access code and an expiry.
// The page itself is static (frontend/index.html), hosted on github.io.
const TENANT='760e00cb-f926-4101-845f-90fd0a72eebf', CLIENT='4fcac966-8904-44ee-be82-9c26fdf23fd2';
const ORIGIN='https://ronitum.github.io';
const MSG={7000215:'Invalid client secret',7000222:'Client secret expired',700016:'Application not found in tenant',90002:'Tenant not found',7000218:'Client assertion or secret missing',invalid_client:'Client authentication failed'};
const H={'Cache-Control':'no-store','X-Content-Type-Options':'nosniff','Access-Control-Allow-Origin':ORIGIN,'Access-Control-Allow-Headers':'content-type,x-access','Access-Control-Allow-Methods':'POST,OPTIONS','Vary':'Origin'};
export default {async fetch(req,env,ctx,doFetch=fetch){
 const u=new URL(req.url);
 const live=env.EXPIRES_AT&&Date.now()<Date.parse(env.EXPIRES_AT)&&env.ACCESS_CODE;
 if(!live||u.pathname!=='/test')return new Response('Not found',{status:404,headers:H});
 if(req.method==='OPTIONS')return new Response(null,{status:204,headers:H});
 if(req.method!=='POST')return new Response('Not found',{status:404,headers:H});
 if(req.headers.get('origin')!==ORIGIN)return Response.json({error:'forbidden'},{status:403,headers:H});
 if(req.headers.get('x-access')!==env.ACCESS_CODE)return Response.json({error:'forbidden'},{status:403,headers:H});
 let secret;try{secret=String((await req.json()).secret||'').trim()}catch{return Response.json({error:'bad request'},{status:400,headers:H})}
 if(!secret||secret.length>200)return Response.json({error:'secret missing or too long'},{status:400,headers:H});
 let r,j;
 try{r=await doFetch(`https://login.microsoftonline.com/${TENANT}/oauth2/v2.0/token`,{method:'POST',headers:{'Content-Type':'application/x-www-form-urlencoded'},body:new URLSearchParams({client_id:CLIENT,client_secret:secret,grant_type:'client_credentials',scope:'https://graph.microsoft.com/.default'}),signal:AbortSignal.timeout(10000)});j=await r.json().catch(()=>({}))}
 catch{return Response.json({ok:false,error:'microsoft_unreachable_or_timeout'},{status:502,headers:H})}
 if(!r.ok){const codes=(Array.isArray(j.error_codes)?j.error_codes:[]).filter(Number.isInteger).slice(0,3);const rid=(r.headers.get('x-ms-request-id')||'').replace(/[^0-9a-f-]/gi,'').slice(0,40);
  return Response.json({ok:false,http:r.status,error:/^[a-z_]{1,40}$/.test(j.error||'')?j.error:'unknown',codes,message:MSG[codes[0]]||MSG[j.error]||'Microsoft rejected the request',requestId:rid},{headers:H})}
 let roles='(unreadable)';try{const p=j.access_token.split('.')[1].replace(/-/g,'+').replace(/_/g,'/');roles=JSON.parse(atob(p+'='.repeat((4-p.length%4)%4))).roles||'(no roles claim)'}catch{}
 return Response.json({ok:true,http:200,expiresIn:j.expires_in,roles},{headers:H}); // token itself is never returned
}};
